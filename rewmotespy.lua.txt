-- ================= SERIALIZER CHUẨN LUA =================
local function serialize(val, depth, indent, seen)
	depth = depth or 0
	indent = indent or 0
	seen = seen or {}
	local pad = string.rep("\t", indent)
	local inner = string.rep("\t", indent + 1)
	local t = typeof(val)

	if t == "string" then
		return string.format("%q", val)
	elseif t == "number" or t == "boolean" then
		return tostring(val)
	elseif t == "nil" then
		return "nil"
	elseif t == "Vector3" then
		return string.format("Vector3.new(%s, %s, %s)", tostring(val.X), tostring(val.Y), tostring(val.Z))
	elseif t == "Vector2" then
		return string.format("Vector2.new(%s, %s)", tostring(val.X), tostring(val.Y))
	elseif t == "CFrame" then
		local c = { val:GetComponents() }
		for i, v in ipairs(c) do c[i] = tostring(v) end
		return "CFrame.new(" .. table.concat(c, ", ") .. ")"
	elseif t == "Color3" then
		return string.format("Color3.fromRGB(%d, %d, %d)", math.floor(val.R * 255 + 0.5), math.floor(val.G * 255 + 0.5), math.floor(val.B * 255 + 0.5))
	elseif t == "UDim2" then
		return string.format("UDim2.new(%s, %s, %s, %s)", tostring(val.X.Scale), tostring(val.X.Offset), tostring(val.Y.Scale), tostring(val.Y.Offset))
	elseif t == "UDim" then
		return string.format("UDim.new(%s, %s)", tostring(val.Scale), tostring(val.Offset))
	elseif t == "EnumItem" then
		return tostring(val)
	elseif t == "Instance" then
		local ok, full = pcall(function() return val:GetFullName() end)
		if ok and type(full) == "string" and full ~= "" then
			local parts = {}
			for p in string.gmatch(full, "[^%.]+") do
				table.insert(parts, p)
			end
			if #parts > 0 then
				local expr = string.format("game:GetService(%q)", parts[1])
				for i = 2, #parts do
					expr = string.format("%s:WaitForChild(%q)", expr, parts[i])
				end
				return expr
			end
		end
		return "nil"
	elseif t == "table" then
		if depth >= 4 or seen[val] then return "{}" end
		seen[val] = true
		local lines = {}
		local count = 0
		local isArr = (#val > 0)
		for k, v in pairs(val) do
			count += 1
			if count > 40 then
				table.insert(lines, inner .. "-- ...")
				break
			end
			local vStr = serialize(v, depth + 1, indent + 1, seen)
			if isArr and type(k) == "number" then
				table.insert(lines, inner .. vStr)
			elseif type(k) == "string" then
				if k:match("^[%a_][%w_]*$") then
					table.insert(lines, string.format("%s%s = %s", inner, k, vStr))
				else
					table.insert(lines, string.format("%s[%q] = %s", inner, k, vStr))
				end
			else
				table.insert(lines, string.format("%s[%s] = %s", inner, serialize(k, depth + 1, 0, seen), vStr))
			end
		end
		seen[val] = nil
		if #lines == 0 then return "{}" end
		return "{\n" .. table.concat(lines, ",\n") .. "\n" .. pad .. "}"
	end

	return string.format("%q", tostring(val))
end

local function getRemotePath(remote)
	local ok, full = pcall(function() return remote:GetFullName() end)
	if not ok or type(full) ~= "string" or full == "" then return "nil" end
	local parts = {}
	for p in string.gmatch(full, "[^%.]+") do
		table.insert(parts, p)
	end
	local expr = string.format("game:GetService(%q)", parts[1])
	for i = 2, #parts do
		expr = string.format("%s:WaitForChild(%q)", expr, parts[i])
	end
	return expr
end

local function generateScript(remote, method, args)
	local path = getRemotePath(remote)
	local callMethod = (method == "InvokeServer") and "InvokeServer" or "FireServer"
	if args and #args > 0 then
		local argLines = {}
		for i = 1, #args do
			table.insert(argLines, "\t" .. serialize(args[i], 0, 1))
		end
		return string.format(
			"local args = {\n%s\n}\n\n%s:%s(unpack(args))",
			table.concat(argLines, ",\n"),
			path,
			callMethod
		)
	end
	return string.format("%s:%s()", path, callMethod)
end

local function copyToClipboard(txt)
	local fn = setclipboard or toclipboard or set_clipboard
	if type(fn) == "function" then
		return pcall(fn, txt)
	end
	return false
end

		if ch:IsA("TextButton") then

		end
	end

	local q = State.Search:lower()
	for i, entry in ipairs(State.Logs) do
		if allowDir(entry.dir) and not State.Excluded[entry.name] then
			local matchStr = (entry.name .. " " .. entry.scriptText):lower()
			if q == "" or matchStr:find(q, 1, true) then
				local blk = State.BlockedNames[entry.name] or (entry.remote and State.BlockedObjs[entry.remote])
				local itemBtn = Instance.new("TextButton")
				itemBtn.Size = UDim2.new(1, -6, 0, 24)

					and Color3.fromRGB(44, 58, 82)
					or (blk and Color3.fromRGB(56, 22, 26) or Color3.fromRGB(22, 26, 33))

					and Color3.fromRGB(190, 65, 75)
					or ((entry.dir == "OUT") and Color3.fromRGB(48, 120, 82) or Color3.fromRGB(70, 85, 110))

local function logRemote(dir, remote, method, args)
	if not State.Enabled then return end
	if typeof(remote) ~= "Instance" then return end
	if not allowDir(dir) then return end

	local name = remote.Name
	if State.Excluded[name] then return end

	local scriptText = generateScript(remote, method, args)
	local last = State.Logs[#State.Logs]
	if last and last.name == name and last.dir == dir and last.scriptText == scriptText then
		last.count += 1
		State.DirtyUI = true
		return
	end

	table.insert(State.Logs, {
		dir = dir,
		name = name,
		remote = remote,
		method = method,
		scriptText = scriptText,
		count = 1,
	})

	if #State.Logs > 250 then
		table.remove(State.Logs, 1)
		if State.Selected then
			State.Selected -= 1
			if State.Selected < 1 then State.Selected = nil end
		end
	end

	State.DirtyUI = true
end

local freezeConn = nil
local function setFreeze(active)
	State.Freeze = active
	if active then
		freezeBtn.Text = "FRZ: ON"
		freezeBtn.BackgroundColor3 = Color3.fromRGB(165, 55, 55)
		statusBox.Text = "Frozen!"
		if not freezeConn then
			freezeConn = RunService.RenderStepped:Connect(function()
				if State.Freeze then
					local t0 = os.clock()
					while State.Freeze and (os.clock() - t0) < 0.045 do end
				end
			end)
		end
	else
		freezeBtn.Text = "FRZ: OFF"
		freezeBtn.BackgroundColor3 = Color3.fromRGB(62, 56, 82)
		statusBox.Text = "Unfrozen"
		if freezeConn then
			freezeConn:Disconnect()
			freezeConn = nil
		end
	end
end

if type(fnHookMeta) == "function" and type(fnGetNamecall) == "function" then
	pcall(function()
		local oldNamecall
		oldNamecall = fnHookMeta(game, "__namecall", wrapClosure(function(self, ...)
			local method = fnGetNamecall()
			if typeof(self) == "Instance" and (method == "FireServer" or method == "InvokeServer") then
				if self:IsA("RemoteEvent") or self:IsA("UnreliableRemoteEvent") or self:IsA("RemoteFunction") then
					if isGameCall() then
						if isBlocked(self) then
							return nil
						end
						local args = { ... }
						task.spawn(logRemote, "OUT", self, method, args)
					end
				end
			end
			return oldNamecall(self, ...)
		end))
	end)
end

if type(fnHookFunc) == "function" then
	pcall(function()
		local tmpEvent = Instance.new("RemoteEvent")
		local oldFire
		oldFire = fnHookFunc(tmpEvent.FireServer, wrapClosure(function(self, ...)
			if typeof(self) == "Instance" and (self:IsA("RemoteEvent") or self:IsA("UnreliableRemoteEvent")) then
				if isGameCall() then
					if isBlocked(self) then
						return nil
					end
					local args = { ... }
					task.spawn(logRemote, "OUT", self, "FireServer", args)
				end
			end
			return oldFire(self, ...)
		end))
		tmpEvent:Destroy()
	end)

	pcall(function()
		local tmpFunc = Instance.new("RemoteFunction")
		local oldInvoke
		oldInvoke = fnHookFunc(tmpFunc.InvokeServer, wrapClosure(function(self, ...)
			if typeof(self) == "Instance" and self:IsA("RemoteFunction") then
				if isGameCall() then
					if isBlocked(self) then
						return nil
					end
					local args = { ... }
					task.spawn(logRemote, "OUT", self, "InvokeServer", args)
				end
			end
			return oldInvoke(self, ...)
		end))
		tmpFunc:Destroy()
	end)
end

local hookedIn = setmetatable({}, { __mode = "k" })
local function attachIn(obj)
	if hookedIn[obj] then return end
	if obj:IsA("RemoteEvent") or obj:IsA("UnreliableRemoteEvent") then
		hookedIn[obj] = true
		obj.OnClientEvent:Connect(function(...)
			logRemote("IN", obj, "OnClientEvent", { ... })
		end)
	end
end

for _, v in ipairs(ReplicatedStorage:GetDescendants()) do
	attachIn(v)
end
ReplicatedStorage.DescendantAdded:Connect(attachIn)
